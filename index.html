Quick start / JavaScript 
// Get all countries
fetch('./api/v2/countries.json')
  .then(function (response) { return response.json(); })
  .then(function (data) { console.log(data); });

// Get a country by ISO2
fetch('./api/v2/countries/us.json')
  .then(function (response) { return response.json(); })
  .then(function (data) { console.log(data); });
